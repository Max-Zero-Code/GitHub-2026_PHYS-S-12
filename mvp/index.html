Goals

  • Finish printing the casing for the sensor ESP32
  • Create a fully fledged network relay system
  • Create a proper menu and touch screen for the screen
  • Print the external casing for the screen
  • Finish protoboard for screen electronics

Explanation

My project's main idea is a forest fire prevention system. Various units of the product are placed in forest areas with vegetation, and using CO2, moisture, temperature, IR, and capacitive sensors, the product is able to monitor the main factors that lead to forest fires. Through a hosted network extended by each Xiao, sensor data is sent to a main console where all statistics can be actively monitored by firefighters.

Sensor Node Code (Xiao — Sender)

#include <WiFi.h>
#include <esp_now.h>
#include "DHT.h"

#define IR_PIN   D0
#define GAS_PIN  D1
#define SOIL_PIN D2
#define DHTPIN   D4
#define DHTTYPE  DHT22

DHT dht(DHTPIN, DHTTYPE);

typedef struct SensorPacket {
  int   irRaw;
  int   gasRaw;
  int   soilRaw;
  float tempC;
  float humidity;
  bool  dhtValid;
} SensorPacket;

SensorPacket packet;
uint8_t broadcastAddress[] = {0xFF, 0xFF, 0xFF, 0xFF, 0xFF, 0xFF};

void onDataSent(const wifi_tx_info_t *tx_info, esp_now_send_status_t status) {
  Serial.print("Send status: ");
  Serial.println(status == ESP_NOW_SEND_SUCCESS ? "OK" : "FAIL");
}

void setup() {
  Serial.begin(115200);
  delay(500);

  analogReadResolution(12);
  dht.begin();

  WiFi.mode(WIFI_STA);

  if (esp_now_init() != ESP_OK) {
    Serial.println("ESP-NOW init failed");
    return;
  }
  esp_now_register_send_cb(onDataSent);

  esp_now_peer_info_t peerInfo = {};
  memcpy(peerInfo.peer_addr, broadcastAddress, 6);
  peerInfo.channel = 0;
  peerInfo.encrypt = false;
  if (esp_now_add_peer(&peerInfo) != ESP_OK) {
    Serial.println("Failed to add peer");
    return;
  }

  Serial.println("Sensor node ready, broadcasting via ESP-NOW.");
}

void loop() {
  packet.irRaw   = analogRead(IR_PIN);
  packet.gasRaw  = analogRead(GAS_PIN);
  packet.soilRaw = analogRead(SOIL_PIN);

  float h = dht.readHumidity();
  float t = dht.readTemperature();
  if (isnan(h) || isnan(t)) {
    packet.dhtValid = false;
  } else {
    packet.dhtValid = true;
    packet.tempC = t;
    packet.humidity = h;
  }

  esp_err_t result = esp_now_send(broadcastAddress, (uint8_t *)&packet, sizeof(packet));

  Serial.print("IR: "); Serial.print(packet.irRaw);
  Serial.print("  Gas: "); Serial.print(packet.gasRaw);
  Serial.print("  Soil: "); Serial.print(packet.soilRaw);
  Serial.print("  DHT: ");
  if (packet.dhtValid) {
    Serial.print(packet.tempC); Serial.print("C "); Serial.print(packet.humidity); Serial.print("%");
  } else {
    Serial.print("invalid");
  }
  Serial.println(result == ESP_OK ? "  [sent]" : "  [send error]");

  delay(2000);
}

Display Host Code (Xiao — Receiver)

#define LGFX_USE_V1
#include <LovyanGFX.hpp>
#include <WiFi.h>
#include <esp_now.h>

class LGFX : public lgfx::LGFX_Device {
  lgfx::Panel_ILI9488 _panel_instance;
  lgfx::Bus_SPI       _bus_instance;

public:
  LGFX(void) {
    auto bus_cfg = _bus_instance.config();
    bus_cfg.spi_host   = SPI2_HOST;
    bus_cfg.spi_mode   = 0;
    bus_cfg.freq_write = 8000000;
    bus_cfg.freq_read  = 4000000;
    bus_cfg.pin_sclk   = D8;
    bus_cfg.pin_mosi   = D10;
    bus_cfg.pin_miso   = D9;
    bus_cfg.pin_dc     = D2;
    _bus_instance.config(bus_cfg);
    _panel_instance.setBus(&_bus_instance);

    auto panel_cfg = _panel_instance.config();
    panel_cfg.pin_cs    = D1;
    panel_cfg.pin_rst   = D3;
    panel_cfg.pin_busy  = -1;
    panel_cfg.panel_width  = 320;
    panel_cfg.panel_height = 480;
    panel_cfg.offset_rotation = 0;
    _panel_instance.config(panel_cfg);

    setPanel(&_panel_instance);
  }
};

LGFX tft;

typedef struct SensorPacket {
  int   irRaw;
  int   gasRaw;
  int   soilRaw;
  float tempC;
  float humidity;
  bool  dhtValid;
} SensorPacket;

SensorPacket lastPacket;
volatile bool newData = false;
unsigned long lastReceivedMs = 0;

void onDataRecv(const esp_now_recv_info_t *recvInfo, const uint8_t *incomingData, int len) {
  if (len != sizeof(SensorPacket)) return;
  memcpy(&lastPacket, incomingData, sizeof(lastPacket));
  newData = true;
  lastReceivedMs = millis();

  Serial.print("IR: "); Serial.print(lastPacket.irRaw);
  Serial.print("  Gas: "); Serial.print(lastPacket.gasRaw);
  Serial.print("  Soil: "); Serial.print(lastPacket.soilRaw);
  Serial.print("  DHT: ");
  if (lastPacket.dhtValid) {
    Serial.print(lastPacket.tempC); Serial.print("C "); Serial.print(lastPacket.humidity); Serial.print("%");
  } else {
    Serial.print("invalid");
  }
  Serial.println("  [received]");
}

void drawReading(int y, const char *label, String value) {
  tft.fillRect(10, y, 300, 24, TFT_BLACK);
  tft.setCursor(10, y);
  tft.setTextColor(TFT_WHITE);
  tft.setTextSize(2);
  tft.print(label);
  tft.print(": ");
  tft.println(value);
}

void setup() {
  Serial.begin(115200);
  delay(500);

  tft.init();
  tft.setRotation(1);
  tft.fillScreen(TFT_RED);
  delay(2000);

  tft.fillScreen(TFT_BLACK);
  tft.setTextColor(TFT_WHITE);
  tft.setTextSize(2);
  tft.setCursor(10, 10);
  tft.println("Waiting for sensor data...");

  WiFi.mode(WIFI_STA);

  if (esp_now_init() != ESP_OK) {
    Serial.println("ESP-NOW init failed");
    return;
  }
  esp_now_register_recv_cb(onDataRecv);

  Serial.println("Display host ready, listening for ESP-NOW packets.");
}

void loop() {
  if (newData) {
    newData = false;
    tft.fillScreen(TFT_BLACK);

    drawReading(20,  "IR raw",   String(lastPacket.irRaw));
    drawReading(50,  "Gas raw",  String(lastPacket.gasRaw));
    drawReading(80,  "Soil raw", String(lastPacket.soilRaw));

    if (lastPacket.dhtValid) {
      drawReading(110, "Temp", String(lastPacket.tempC, 1) + "C");
      drawReading(140, "Humidity", String(lastPacket.humidity, 1) + "%");
    } else {
      drawReading(110, "DHT", "read failed");
    }
  }

  if (millis() - lastReceivedMs > 10000 && lastReceivedMs != 0) {
    tft.fillRect(10, 180, 300, 24, TFT_BLACK);
    tft.setCursor(10, 180);
    tft.setTextColor(TFT_RED);
    tft.setTextSize(2);
    tft.println("Signal lost");
  }
}
